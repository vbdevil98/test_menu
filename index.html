const express = require('express'); const Groq = require('groq-sdk'); const path = require('path'); const app = express(); app.use(express.json()); // Essential for parsing incoming data // SECURELY INITIALIZE GROQ // This only works because we are running Node.js on Render's backend const groq = new Groq({ apiKey: process.env.GROQ_API_KEY || "WARNING_KEY_NOT_FOUND" }); // Serve static frontend files (the UI) from the root directory app.use(express.static(path.join(__dirname, '/'))); // ========================================== // BACKEND API ENDPOINT: SECURE AI SOMMELIER // ========================================== app.post('/generateAISummary', async (req, res) => { // SECURITY: The client only sends the item name and description. // The actual key and AI call are handled privately on Render's server. const { itemName, itemDescription } = req.body; if (!process.env.GROQ_API_KEY) { return res.status(500).json({ error: "Backend Config Error: GROQ_API_KEY environment variable missing on Render." }); } try { const chatCompletion = await groq.chat.completions.create({ messages: [{ role: "user", content: `Analyze this menu item for a trendy Gen-Z restaurant called 'Tattva'. In exactly 2 sentences, provide a high-vibe summary focusing on: nutritional high points (e.g., 'High Protein', 'Keto', 'Plant-Powered'), a brief sensory description, and one standard menu pairing to upsell. Use one relevant emoji. Item: "${itemName}" - "${itemDescription}"` }], model: "llama3-8b-8192", // Fast and efficient model temperature: 0.7 }); const summary = chatCompletion.choices[0].message.content; res.json({ summary }); } catch (error) { console.error("GROQ API Error:", error); res.status(500).json({ error: "Unable to reach the AI Sommelier at this moment." }); } }); // Default to port 10000 or Render's PORT variable const PORT = process.env.PORT || 10000; app.listen(PORT, () => { console.log(`Backend Service listening on port ${PORT}`); }); // ========================================== // FRONTEND BROWSER CODE (UI) // ========================================== const html_content = ` Tattva Menu | AI Sommelier

Tattva

Curated. Conscious. Crafted.

AI Sommelier

Summary generated privately on Render backend.

\`; // Define a final route to serve the internal HTML content app.get('/', (req, res) => { res.send(html_content); });
